# ==========================================================================
# DSLR - Dynamic Sign Language Recognition System
# requirements.txt
#
# NOTE: Some packages differ between the Colab training environment (GPU,
# full TensorFlow) and the Raspberry Pi inference environment (ARM, CPU-only,
# lightweight). Where relevant this is called out below. If you want strict
# separation, use requirements-train.txt / requirements-pi.txt instead
# (see bottom of this file for the split).
# ==========================================================================

# --- Core ML / CV stack ---
tensorflow          # Colab: full TF. On Pi, consider tflite-runtime instead (see below)
mediapipe           # Landmark extraction (Holistic / Hands) - works on both Colab & Pi
opencv-python    # Video capture + frame processing
numpy
pandas
scikit-learn         # train_test_split, metrics, label encoding

# --- Visualization / training utilities (Colab side mainly) ---
matplotlib
seaborn
tqdm

# --- Text-to-Speech (Raspberry Pi inference side) ---
pyttsx3                # Offline, lightweight TTS (espeak backend on Linux/Pi)

# --- Misc utilities ---
gTTS                  # Optional: online fallback TTS with nicer voices (needs internet)

# ==========================================================================
# SPLIT INSTALL GUIDE (recommended for real deployments)
# ==========================================================================
# Colab / training machine:
#   pip install tensorflow mediapipe opencv-python numpy pandas scikit-learn matplotlib seaborn tqdm
#
# Raspberry Pi (3B+/4/5) - lighter footprint, use TFLite runtime instead of full TF:
#   pip install mediapipe opencv-python numpy pyttsx3
#   pip install tflite-runtime            # ARM wheel, much smaller & faster than full tensorflow
#   sudo apt-get install espeak libespeak1  # pyttsx3 dependency on Linux/Raspberry Pi OS
# ==========================================================================
